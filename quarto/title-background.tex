%% Preamble for Quarto Beamer decks (include-in-header): the JECS / NIES title
%% background on page 1, the title page written by before-body.tex.
%%
%% The themes' \titleframe sets the background canvas just before its frame,
%% but in a Pandoc deck (ignorenonframetext) anything between frames is
%% skipped, and a setting inside the frame comes too late. So the canvas is
%% replaced once, in the preamble: page 1 gets the title image instead of the
%% theme's canvas (the title image has transparent parts, so drawing it on top
%% of the content banner would let the banner show through); every other
%% page keeps the theme's canvas. Image names and placement are copied from
%% the themes' \titleframe; keep them in step.
\makeatletter
\AtBeginDocument{%
  \expandafter\let\expandafter\ib@themecanvas
    \csname beamer@@tmpl@background canvas\endcsname
  \ifcsname ver@beamerouterthemeJECS.sty\endcsname
    \setbeamertemplate{background canvas}{%
      \ifnum\c@page=1
        \tikz[remember picture,overlay] \node[anchor=north west, inner sep=0pt]
          at (current page.north west)
          {\includegraphics[width=\paperwidth,height=\paperheight]{images/title-bg.png}};%
      \else\ib@themecanvas\fi}%
  \fi
  \ifcsname ver@beamerouterthemeNIES.sty\endcsname
    \setbeamertemplate{background canvas}{%
      \ifnum\c@page=1
        \tikz[remember picture,overlay] \node[anchor=north east, inner sep=0pt]
          at (current page.north east)
          {\includegraphics[height=\paperheight]{images/nies-title.png}};%
      \else\ib@themecanvas\fi}%
  \fi}
\makeatother
